public class FFMpegMergeCaller {

    public static void concatKeyframeCopyNoReencode(Main.VideoMergeDescription videoMergeDescription,
                                                    Main.InputArguments inputArguments)
            throws IOException, InterruptedException {

        final var clips = videoMergeDescription.descriptionList();
        final var outputFile = inputArguments.outputPath().resolve(videoMergeDescription.outputVideoName());

        Objects.requireNonNull(clips, "clips");
        if (clips.isEmpty()) throw new IllegalArgumentException("clips is empty");

        Path tmpDir = Files.createTempDirectory("ffm-concat-");
        List<Path> cutFiles = new ArrayList<>();
        Path listFile = tmpDir.resolve("list.txt");

        try (BufferedWriter listWriter = Files.newBufferedWriter(listFile, StandardCharsets.UTF_8)) {
            for (int i = 0; i < clips.size(); i++) {
                Main.VideoDescription vd = clips.get(i);
                Path src = Paths.get(vd.videoName());
                if (!Files.exists(src)) throw new FileNotFoundException("Not found: " + src);

                // если нужно "повернуть" → сначала делаем временную копию с rotate=180
                boolean needRotate = vd.options().contains(Main.ProcessingOptions.R);
                Path effective = src;
                if (needRotate) {
                    Path rotated = tmpDir.resolve("rotated_" + i + ".mp4");
                    List<String> rotateCmd = List.of(
                            "ffmpeg", "-y",
                            "-i", src.toString(),
                            "-map", "0:v:0", "-map", "0:a?",
                            "-c", "copy",
                            "-metadata:s:v:0", "rotate=180",
                            rotated.toString()
                    );
                    runCommand(rotateCmd);
                    effective = rotated;
                }

                // находим ближайшие keyframe-границы
                Optional<List<Double>> keysOpt = Optional.of(getKeyframes(effective)).filter(list -> !list.isEmpty());
                double inRaw = toSeconds(vd.start());
                double inSnap = inRaw;
                Double outSnap = "end".equalsIgnoreCase(vd.end()) ? null : toSeconds(vd.end());

                if (keysOpt.isPresent()) {
                    List<Double> keys = keysOpt.get();
                    inSnap = snapPrevKey(keys, inRaw);
                    if (outSnap != null) {
                        double outRaw = outSnap;
                        outSnap = snapPrevKey(keys, outRaw);
                        if (outSnap - inSnap <= EPS) {
                            Double next = nextKeyAfter(keys, inSnap + EPS);
                            if (next != null && next <= outRaw + EPS) outSnap = next;
                            else outSnap = null; // до конца файла
                        }
                    }
                }

                // режем сегмент в отдельный файл
                Path cutFile = tmpDir.resolve("cut_" + i + ".mp4");
                List<String> cutCmd = new ArrayList<>();
                cutCmd.add("ffmpeg"); cutCmd.add("-y");
                cutCmd.add("-ss"); cutCmd.add(trimTrailingZeros(inSnap));
                if (outSnap != null) {
                    cutCmd.add("-to"); cutCmd.add(trimTrailingZeros(outSnap));
                }
                cutCmd.add("-i"); cutCmd.add(effective.toString());
                cutCmd.add("-map"); cutCmd.add("0:v:0");
                cutCmd.add("-map"); cutCmd.add("0:a?");
                cutCmd.add("-c"); cutCmd.add("copy");
                cutCmd.add(cutFile.toString());
                runCommand(cutCmd);

                cutFiles.add(cutFile);
                listWriter.write("file '" + cutFile.toAbsolutePath().toString().replace("\\", "/") + "'\n");
            }
        }

        // финальная склейка уже готовых сегментов
        List<String> concatCmd = List.of(
                "ffmpeg", "-y",
                "-f", "concat",
                "-safe", "0",
                "-i", listFile.toString(),
                "-c", "copy",
                outputFile.toString()
        );
        runCommand(concatCmd);

        System.out.println("Done: " + outputFile.toAbsolutePath());

        // очистка временных файлов
        try (var s = Files.walk(tmpDir)) {
            s.sorted(Comparator.reverseOrder()).forEach(p -> {
                try { Files.deleteIfExists(p); } catch (Exception ignored) {}
            });
        }
    }

    /** Один сегмент в итоговом списке: файл + in/out в секундах (out=null => до конца файла). */
    private record Segment(Path file, double inSec, Double outSec) {}

    /** Запись ffconcat файла версии 1.0, с inpoint/outpoint. Пути приводим к абсолютным и с прямыми слешами. */
    private static void writeFfconcat(Path concatFile, List<Segment> segments) throws IOException {
        StringBuilder sb = new StringBuilder();
        sb.append("ffconcat version 1.0\n");
        for (Segment s : segments) {
            String p = s.file.toAbsolutePath().toString().replace('\\', '/');
            sb.append("file ").append(quoteIfNeeded(p)).append("\n");
            sb.append("inpoint ").append(trimTrailingZeros(s.inSec)).append("\n");
            if (s.outSec != null) {
                sb.append("outpoint ").append(trimTrailingZeros(s.outSec)).append("\n");
            }
            if (s.outSec != null) {
                double dur = s.outSec - s.inSec;
                sb.append("duration ").append(trimTrailingZeros(dur)).append("\n");
            }
        }
        Files.writeString(concatFile, sb.toString(), StandardCharsets.UTF_8);
    }

    private static void runCommand(List<String> cmd) throws IOException, InterruptedException {
       SimpleCommandRunner.i().runCommand(cmd);
    }

    /** HH:MM:SS -> секунды (double). */
    private static double toSeconds(String hhmmss) {
        String[] t = hhmmss.split(":");
        int h = Integer.parseInt(t[0]);
        int m = Integer.parseInt(t[1]);
        int s = Integer.parseInt(t[2]);
        return h * 3600.0 + m * 60.0 + s;
    }

    /** Обрезает .0 у целых чисел для читаемости. */
    private static String trimTrailingZeros(double v) {
        if (Math.abs(v - Math.rint(v)) < 1e-9) return String.valueOf((long) Math.rint(v));
        return String.valueOf(v);
    }

    /** Кавычки для путей с пробелами. */
    private static String quoteIfNeeded(String path) {
        if (path.contains(" ") || path.contains("'")) {
            return "'" + path.replace("'", "'\\''") + "'";
        }
        return path;
    }


    private static final Map<Path, List<Double>> KEYFRAME_CACHE = new HashMap<>();

    /** Возвращает отсортированный список времен (в секундах) всех keyframe’ов видео-дорожки v:0. */
    private static List<Double> getKeyframes(Path file) throws IOException, InterruptedException {
        Path abs = file.toAbsolutePath();
        List<Double> cached = KEYFRAME_CACHE.get(abs);
        if (cached != null) return cached;

        // ffprobe вернет ТОЛЬКО ключевые кадры (skip_frame nokey).
        // CSV без заголовков (p=0): строки вида "0.000000" или с префиксом "frame,0.000000" в старых сборках.
        List<String> cmd = List.of(
                "ffprobe",
                "-v", "error",
                "-select_streams", "v:0",
                "-skip_frame", "nokey",
                "-show_frames",
                "-show_entries", "frame=best_effort_timestamp_time",
                "-of", "csv=p=0",             // без префиксов и заголовков
                abs.toString()
        );

        ProcessBuilder pb = new ProcessBuilder(cmd);
        pb.redirectErrorStream(true);
        Process p = pb.start();

        List<Double> keyframes = new ArrayList<>();
        try (BufferedReader r = new BufferedReader(new InputStreamReader(p.getInputStream(), StandardCharsets.UTF_8))) {
            String line;
            while ((line = r.readLine()) != null) {
                line = line.trim();
                if (line.isEmpty()) continue;

    //            System.out.println(line);
                // На всякий случай поддержим оба варианта вывода:
                // "0.000000"  ИЛИ  "frame,0.000000"
                String val = line.startsWith("frame,") ? line.substring("frame,".length()) : line;

                try {
                    keyframes.add(Double.parseDouble(val));
                } catch (NumberFormatException ignore) {
                    // пропускаем странные строки
                }
            }
        }
        int code = p.waitFor();
        if (code != 0) {
            throw new IOException("ffprobe failed (" + code + "): " + String.join(" ", cmd));
        }

        // На всякий — добавим 0.0 как "первый ключ", если пусто
        if (keyframes.isEmpty() || keyframes.get(0) > 1e-9) {
            keyframes.add(0, 0.0);
        } else {
            // гарантируем сортировку
            Collections.sort(keyframes);
        }

        KEYFRAME_CACHE.put(abs, keyframes);
        return keyframes;
    }

    private static final double EPS = 1e-6;

    /** Возвращает наибольший keyframe <= t. Если все keyframe > t, вернет самый ранний (обычно 0.0). */
    private static double snapPrevKey(List<Double> keys, double t) {
        int i = Collections.binarySearch(keys, t);
        if (i >= 0) {
            // попали ровно в keyframe
            return keys.get(i);
        } else {
            int ins = -i - 1; // первая позиция > t
            if (ins <= 0) return keys.get(0);
            return keys.get(ins - 1);
        }
    }

    /** Возвращает наименьший keyframe > t, или null, если такого нет. */
    private static Double nextKeyAfter(List<Double> keys, double t) {
        int i = Collections.binarySearch(keys, t);
        int ins = (i >= 0) ? i + 1 : -i - 1; // первая позиция > t
        if (ins >= keys.size()) return null;
        return keys.get(ins);
    }

    private static Segment createSegment(Path effective, Main.VideoDescription vd) throws IOException, InterruptedException {
        List<Double> keys = getKeyframes(effective);

        double inRaw = toSeconds(vd.start());
        double inSnap = snapPrevKey(keys, inRaw);

        Double outSnap = null;
        if (!"end".equalsIgnoreCase(vd.end())) {
            double outRaw = toSeconds(vd.end());
            outSnap = snapPrevKey(keys, outRaw);
            // Гарантируем, что out > in. Если нет, пробуем следующий keyframe.
            if (outSnap - inSnap <= EPS) {
                Double next = nextKeyAfter(keys, inSnap + EPS);
                if (next != null && next - inSnap > EPS && next <= outRaw + EPS) {
                    outSnap = next;
                } else {
                    // как fallback — до конца файла
                    outSnap = null;
                }
            }
        }

        return new Segment(effective, inSnap, outSnap);
    }
}
